cmake_minimum_required(VERSION 3.22)
project(aura_mobile_voice LANGUAGES C CXX)

# Android's sideload build stays debuggable, but Qwen and GGML run with release optimization.
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS aura_compile=4)
set(CMAKE_JOB_POOL_COMPILE aura_compile)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build the Qwen runtime statically" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Cross-compile for Android" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "Use GGML worker threads" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_CPU_ARM_ARCH "${AURA_ARM_ARCH}" CACHE STRING "ARM64 CPU profile" FORCE)
elseif(ANDROID_ABI STREQUAL "x86_64")
    set(GGML_AVX OFF CACHE BOOL "Support emulator CPUs" FORCE)
    set(GGML_AVX2 OFF CACHE BOOL "Support emulator CPUs" FORCE)
    set(GGML_FMA OFF CACHE BOOL "Support emulator CPUs" FORCE)
    set(GGML_F16C OFF CACHE BOOL "Support emulator CPUs" FORCE)
endif()

add_subdirectory("${QWEN3_TTS_DIR}" qwen3-tts EXCLUDE_FROM_ALL)
add_library(aura_voice SHARED voice.cpp)
target_link_libraries(aura_voice PRIVATE qwen3_tts)
target_link_options(aura_voice PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
